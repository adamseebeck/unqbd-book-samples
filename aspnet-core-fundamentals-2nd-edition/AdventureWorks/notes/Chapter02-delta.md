### Step 3: Implement the services and register their lifetimes

Create `AdventureWorks.Infrastructure/Services.cs` (32 lines).

```csharp
using System.Collections.Concurrent;
using System.Net.Mail;
using AdventureWorks.Core;

namespace AdventureWorks.Infrastructure;

public sealed class OrderRepository : IOrderRepository
{
    public Order? GetById(int id) =>
        id == 1 ? new Order(1, "AW-0001", 49.00m) : null;
}

public sealed class MemoryCacheService : ICacheService
{
    private readonly ConcurrentDictionary<string, string> _values
        = new();

    public string? Get(string key) =>
        _values.TryGetValue(key, out var value) ? value : null;

    public void Set(string key, string value) =>
        _values[key] = value;
}

public sealed class EmailValidator : IEmailValidator
{
    public bool IsValid(string address) =>
        !string.IsNullOrWhiteSpace(address) &&
        MailAddress.TryCreate(address, out var parsed) &&
        string.Equals(parsed.Address, address,
            StringComparison.OrdinalIgnoreCase);
}
```

app.MapGet("/api/orders/{id:int}", (int id,
    IOrderRepository orders, IOptions<FeatureFlags> flags) =>
{
    if (!flags.Value.EnableNewCheckout)
        return Results.StatusCode(
            StatusCodes.Status503ServiceUnavailable);
    var order = orders.GetById(id);
    return order is null ? Results.NotFound() : Results.Ok(order);
});
if (app.Environment.IsDevelopment())
{
    app.MapGet("/api/config", (IOptions<FeatureFlags> flags) =>
        Results.Ok(flags.Value));
    app.MapGet("/api/email-check",
        (string address, IEmailValidator validator) => Results.Ok(

}
